# Функция барои сохтани калиди тасодуфӣ
def generate_zolo_key(days: int):
    if days == 7:
        duration_str = "7DTrial"
    elif days == 60:
        duration_str = "60DTrial"
    else:
        duration_str = f"{days}DTrial"
        
    letters_and_digits = string.ascii_letters + string.digits
    random_part = ''.join(random.choice(letters_and_digits) for _ in range(16))
    
    key = f"ZOLO-{duration_str}-{random_part}"
    return key

# Менюи тугмаҳо
def get_main_menu():
    keyboard = types.InlineKeyboardMarkup(row_width=1)
    btn1 = types.InlineKeyboardButton("🔑 Гирифтани ключи 7 рӯза", callback_data="get_7d")
    btn2 = types.InlineKeyboardButton("🔑 Гирифтани ключи 60 рӯза", callback_data="get_60d")
    keyboard.add(btn1, btn2)
    return keyboard

# Фармони /start
@bot.message_handler(commands=['start'])
def send_welcome(message):
    welcome_text = (
        f"Салом, **{message.from_user.first_name}**! 👋\n\n"
        "Ин боти ройгони тавлиди ключҳои **Zolocheat** аст.\n"
        "Барои гирифтани ключ тугмаи лозимиро зер пахш кунед:"
    )
    bot.send_message(message.chat.id, welcome_text, parse_mode="Markdown", reply_markup=get_main_menu())

# Коркарди тугмаҳо (Callback queries)
@bot.callback_query_handler(func=lambda call: True)
def callback_query(call):
    if call.data == 'get_7d':
        days = 7
    elif call.data == 'get_60d':
        days = 60
    else:
        return

    # Тавлиди ключ
    new_key = generate_zolo_key(days)
    
    response_text = (
        f"✅ **Ключи шумо омода аст!**\n\n"
        f"`{new_key}`\n\n"
        f"Мӯҳлат: **{days} рӯз**\n"
        f"⚠️ *Эзоҳ: Ин ключҳо ройгон ва тасодуфӣ тавлид шудаанд.*"
    )
    
    # Фиристодани ҷавоб бо навсозии паём ё фиристодани паёми нав
    bot.answer_callback_query(call.id)
    bot.send_message(call.message.chat.id, response_text, parse_mode="Markdown", reply_markup=get_main_menu())

if __name__ == '__main__':
